{%- comment -%} Numbered pagination. Every page is a real file written at build time (/blog/page/2/ and so on), so this works on GitHub Pages exactly as it works locally — no JavaScript, no query strings, no server. The window of numbers comes from `pagination.trail` in _config.yml; the first and last are always reachable, with an ellipsis where numbers are skipped. {%- endcomment -%} {%- if paginator.total_pages > 1 -%} {%- assign trail = paginator.page_trail -%} {%- assign first_shown = trail | first -%} {%- assign last_shown = trail | last -%}
{%- if paginator.previous_page -%} Newer {%- else -%} Newer {%- endif -%}
    {%- if first_shown.num > 1 -%}
  1. 1
    2. {%- if first_shown.num > 2 %}
  2. …
    3. {% endif -%} {%- endif -%} {%- for t in trail -%}
  3. {%- if t.num == paginator.page -%} {{ t.num }} {%- else -%} {{ t.num }} {%- endif -%}
    4. {%- endfor -%} {%- if last_shown.num < paginator.total_pages -%} {%- assign gap = paginator.total_pages | minus: 1 -%} {%- if last_shown.num < gap %}
  4. …
    5. {% endif -%}
  5. {{ paginator.total_pages }}
    6. {%- endif -%}
{%- if paginator.next_page -%} Older {%- else -%} Older {%- endif -%}
{%- endif -%}
